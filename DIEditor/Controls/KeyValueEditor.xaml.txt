<?xml version="1.0" encoding="utf-8"?>
<UserControl
    x:Class="DIEditor.Controls.KeyValueEditor"
    xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
    xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
    xmlns:local="using:DIEditor.Controls"
    xmlns:d="http://schemas.microsoft.com/expression/blend/2008"
    xmlns:mc="http://schemas.openxmlformats.org/markup-compatibility/2006"
    mc:Ignorable="d">

    <UserControl.Resources>
        <DataTemplate x:Key="KeyValueRowTemplate" x:DataType="local:KeyValueItem">

            <!-- 每行外框：圆角 + 淡描边 + 卡片底色，让相邻条目在视觉上分开 -->
            <Border Margin="0,0,0,8"
                    Padding="8"
                    CornerRadius="6"
                    BorderThickness="1"
                    BorderBrush="{ThemeResource CardStrokeColorDefaultBrush}"
                    Background="{ThemeResource CardBackgroundFillColorDefaultBrush}">

                <!-- 五列：浏览 | 键 | 竖线 | 值 | 删除；另有一行专门显示校验错误 -->
                <Grid ColumnSpacing="8" RowSpacing="4">

                    <Grid.ColumnDefinitions>
                        <!-- 0：浏览按钮，宽度随文字自适应 -->
                        <ColumnDefinition Width="Auto" />
                        <!-- 1：左输入框（键），星号宽度参与伸缩 -->
                        <ColumnDefinition Width="*" MinWidth="90" />
                        <!-- 2：中间分割线，固定宽度不参与伸缩 -->
                        <ColumnDefinition Width="Auto" />
                        <!-- 3：右输入框（值） -->
                        <ColumnDefinition Width="*" MinWidth="90" />
                        <!-- 4：最右侧删除按钮 -->
                        <ColumnDefinition Width="Auto" />
                    </Grid.ColumnDefinitions>

                    <Grid.RowDefinitions>
                        <RowDefinition Height="Auto" />
                        <RowDefinition Height="Auto" />
                    </Grid.RowDefinitions>

                    <!-- ==================== 最左：浏览文件 ==================== -->
                    <!-- Tag 带出本行数据对象，处理器据此知道“填到哪一行”；
                         按钮高度跟随同行 TextBox，VerticalAlignment=Stretch 让两者等高 -->
                    <Button Grid.Row="0" Grid.Column="0"
                            Height="32"
                            Padding="10,0"
                            VerticalAlignment="Center"
                            Click="BrowseItem_Click"
                            Tag="{x:Bind}"
                            ToolTipService.ToolTip="从文件中选择，仅取文件名填入键"
                            AutomationProperties.Name="选择文件">
                        <StackPanel Orientation="Horizontal" Spacing="6">
                            <FontIcon Glyph="&#xE8E5;" FontSize="13" />
                            <TextBlock Text="浏览..." FontSize="13" />
                        </StackPanel>
                    </Button>

                    <!-- ==================== 左：键输入框 ==================== -->
                    <!-- UpdateSourceTrigger=PropertyChanged：x:Bind 对 TextBox.Text 默认是
                         LostFocus，不写这行会变成“失焦才回写”，浏览填充后的联动也会迟一拍 -->
                    <TextBox Grid.Row="0" Grid.Column="1"
                             PlaceholderText="原始文件名"
                             AutomationProperties.Name="键"
                             Text="{x:Bind Key, Mode=TwoWay, UpdateSourceTrigger=PropertyChanged}" />

                    <!-- ==================== 中：1px 竖线 ==================== -->
                    <Border Grid.Row="0" Grid.Column="2"
                            Width="1"
                            Margin="0,4"
                            Background="{ThemeResource DividerStrokeColorDefaultBrush}" />

                    <!-- ==================== 右：值输入框 ==================== -->
                    <TextBox Grid.Row="0" Grid.Column="3"
                             PlaceholderText="自定义文件名"
                             AutomationProperties.Name="值"
                             Text="{x:Bind Value, Mode=TwoWay, UpdateSourceTrigger=PropertyChanged}" />

                    <!-- ==================== 最右：删除本行 ==================== -->
                    <Button Grid.Row="0" Grid.Column="4"
                            Width="36" Height="32" Padding="0"
                            Click="DeleteItem_Click"
                            Tag="{x:Bind}"
                            ToolTipService.ToolTip="删除此条目"
                            AutomationProperties.Name="删除此条目">
                        <FontIcon Glyph="&#xE74D;" FontSize="14" />
                    </Button>

                    <!-- ==================== 校验提示 ==================== -->
                    <TextBlock Grid.Row="1" Grid.Column="1" Grid.ColumnSpan="3"
                               Text="{x:Bind ValidationMessage, Mode=OneWay}"
                               Visibility="{x:Bind ErrorVisibility, Mode=OneWay}"
                               Foreground="{ThemeResource SystemFillColorCriticalBrush}"
                               Style="{StaticResource CaptionTextBlockStyle}"
                               TextWrapping="Wrap" />

                </Grid>
            </Border>
        </DataTemplate>
    </UserControl.Resources>

    <!-- 根布局：上面列表，下面“列表之外”的操作区 -->
    <Grid RowSpacing="8" MaxHeight="250">
        <Grid.RowDefinitions>
            <RowDefinition Height="*" />
            <RowDefinition Height="Auto" />
        </Grid.RowDefinitions>

        <!-- 外部需给本控件一个确定高度（Height 或放在 * 行），ScrollViewer 才会真正滚动 -->
        <ScrollViewer Grid.Row="0"
                      VerticalScrollBarVisibility="Auto"
                      VerticalScrollMode="Auto"
                      HorizontalScrollMode="Disabled"
                      HorizontalScrollBarVisibility="Disabled"
                      Padding="0,0,4,0">

            <ItemsControl x:Name="RowItemsControl"
                          ItemsSource="{x:Bind Items, Mode=OneWay}"
                          ItemTemplate="{StaticResource KeyValueRowTemplate}">

                <!-- ItemsControl 默认不横向拉伸容器，不写这段每行只有内容那么宽 -->
                <ItemsControl.ItemContainerStyle>
                    <Style TargetType="ContentPresenter">
                        <Setter Property="HorizontalAlignment" Value="Stretch" />
                        <Setter Property="HorizontalContentAlignment" Value="Stretch" />
                    </Style>
                </ItemsControl.ItemContainerStyle>

                <ItemsControl.ItemsPanel>
                    <ItemsPanelTemplate>
                        <StackPanel Orientation="Vertical" />
                    </ItemsPanelTemplate>
                </ItemsControl.ItemsPanel>
            </ItemsControl>
        </ScrollViewer>

        <!-- ==================== 列表之外的按钮区 ==================== -->
        <Grid Grid.Row="1" ColumnSpacing="8">
            <Grid.ColumnDefinitions>
                <ColumnDefinition Width="Auto" />
                <ColumnDefinition Width="*" />
            </Grid.ColumnDefinitions>

            <Button Grid.Column="0" Click="AddItem_Click">
                <StackPanel Orientation="Horizontal" Spacing="8">
                    <FontIcon Glyph="&#xE710;" FontSize="14" />
                    <TextBlock Text="添加条目" />
                </StackPanel>
            </Button>

            <TextBlock x:Name="StatusTextBlock"
                       Grid.Column="1"
                       VerticalAlignment="Center"
                       HorizontalAlignment="Right"
                       Style="{StaticResource CaptionTextBlockStyle}"
                       Foreground="{ThemeResource TextFillColorSecondaryBrush}" />
        </Grid>

    </Grid>
</UserControl>
